```bash
cd para
docker compose up -d --build          # сборка и запуск контура (сеет фикстуры)
curl -s -o /dev/null -w '%{http_code}\n' localhost:8080/healthz   # 200, «ok»

# смок-цепочка (ручной вариант; в прогоне сессии среда не имела доступа к порту — HTTP 0)

curl -s localhost:8080/login | grep -c 'static/css/tokens.css'                   # ≥1 — оформление подключено
curl -s -o /dev/null -w '%{http_code} %{redirect_url}\n' localhost:8080/         # без cookie → 303 на /login
curl -s -o /dev/null -w '%{http_code}\n' -X POST localhost:8080/v1/auth/login \
  -H 'Content-Type: application/json' \
  -d '{"login":"anya.demo","password":"заведомо-неверный-пароль"}'               # 401 invalid_credentials

# вход и дашборд (<DEMO_PASSWORD> — единый демо-пароль, публикуется в комментарии

curl -s -b c.txt localhost:8080/               # 200: data-page="dashboard", topbar/tabbar, aria-current на «Сегодня»
curl -s -o /dev/null -w '%{http_code}\n' localhost:8080/static/css/tokens.css   # 200
```
